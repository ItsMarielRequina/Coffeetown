-- Table: staff
CREATE TABLE staff (
    id INT IDENTITY(1,1) PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    email VARCHAR(100) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    created_at DATETIME NOT NULL DEFAULT GETDATE()
);
GO

-- Table: customers
CREATE TABLE customers (
    customerID INT IDENTITY(1,1) PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    contact_number VARCHAR(15) NOT NULL,
    created_at DATETIME NOT NULL DEFAULT GETDATE()
);
GO

-- Table: cart
CREATE TABLE cart (
    id INT IDENTITY(1,1) PRIMARY KEY,
    customer_id INT NULL,
    item_name VARCHAR(255) NULL,
    quantity INT NULL,
    price DECIMAL(10,2) NULL,
    added_at DATETIME DEFAULT GETDATE()
);
GO

-- Table: categorized_expenses
CREATE TABLE categorized_expenses (
    id INT IDENTITY(1,1) PRIMARY KEY,
    category VARCHAR(50) NULL,
    total_cost DECIMAL(10,2) NULL,
    order_date DATE NULL
);
GO

-- Table: categorized_purchases
CREATE TABLE categorized_purchases (
    id INT IDENTITY(1,1) PRIMARY KEY,
    category VARCHAR(50) NULL,
    total_amount DECIMAL(10,2) NULL,
    purchase_date DATE NULL
);
GO

-- Table: financial_summary
CREATE TABLE financial_summary (
    id INT IDENTITY(1,1) PRIMARY KEY,
    summary_date DATE NOT NULL,
    category VARCHAR(20) NOT NULL CHECK (category IN ('daily', 'weekly', 'monthly', 'yearly')),
    summary_name VARCHAR(50) NOT NULL,
    total_sales DECIMAL(10,2) NOT NULL,
    total_expenses DECIMAL(10,2) NOT NULL,
    created_at DATETIME NOT NULL DEFAULT GETDATE()
);
GO

-- Table: products
CREATE TABLE products (
    id INT IDENTITY(1,1) PRIMARY KEY,
    product_name VARCHAR(255) NULL,
    price DECIMAL(10,2) NULL,
    type VARCHAR(50) NULL,
    description VARCHAR(255) NULL,
    images VARCHAR(255) NULL,
    quantity INT NULL
);
GO

-- Table: inventory
CREATE TABLE inventory (
    id INT IDENTITY(1,1) PRIMARY KEY,
    product_name VARCHAR(100) NOT NULL,
    raw_material_cost DECIMAL(10,2) NOT NULL,
    labor_cost DECIMAL(10,2) NOT NULL,
    overhead_cost DECIMAL(10,2) NOT NULL,
    packaging_cost DECIMAL(10,2) NOT NULL,
    total_cost AS (raw_material_cost + labor_cost + overhead_cost + packaging_cost) PERSISTED,
    price DECIMAL(10,2) NOT NULL,
    quantity INT DEFAULT 50,
    stock_supply INT DEFAULT 100,
    supplier_id INT NULL,
    is_new BIT DEFAULT 0
);
GO

-- Table: suppliers
CREATE TABLE suppliers (
    id INT IDENTITY(1,1) PRIMARY KEY,
    product_id INT NOT NULL,
    supplier_name VARCHAR(100) NOT NULL,
    stock_supply INT DEFAULT 100,
    CONSTRAINT FK_suppliers_inventory FOREIGN KEY (product_id) 
        REFERENCES inventory(id) ON DELETE CASCADE
);
GO

-- Table: expenses
CREATE TABLE expenses (
    product_id INT NOT NULL,
    order_quantity INT NOT NULL,
    total_cost DECIMAL(10,2) NOT NULL,
    order_date DATETIME NOT NULL DEFAULT GETDATE(),
    id INT NULL,
    supplier_id INT NULL
);
GO

-- Table: purchases
CREATE TABLE purchases (
    purchaseID INT IDENTITY(1,1) PRIMARY KEY,
    customerID INT NOT NULL,
    item_name VARCHAR(255) NOT NULL,
    quantity INT NOT NULL,
    purchase_date DATETIME DEFAULT GETDATE(),
    price DECIMAL(10,2) NOT NULL,
    total_amount DECIMAL(10,2) NOT NULL
);
GO

-- Table: payment
CREATE TABLE payment (
    paymentID INT IDENTITY(1,1) PRIMARY KEY,
    amount_pay DECIMAL(10,2) NOT NULL,
    paymentDate DATE NOT NULL,
    purchaseID INT NULL,
    CONSTRAINT FK_payment_purchases FOREIGN KEY (purchaseID) 
        REFERENCES purchases(purchaseID)
);
GO

-- Table: receipts
CREATE TABLE receipts (
    id INT IDENTITY(1,1) PRIMARY KEY,
    customerID INT NOT NULL,
    item_name VARCHAR(255) NOT NULL,
    quantity INT NOT NULL,
    price DECIMAL(10,2) NOT NULL,
    total_amount DECIMAL(10,2) NOT NULL,
    order_date DATETIME NOT NULL DEFAULT GETDATE(),
    CONSTRAINT FK_receipts_customers FOREIGN KEY (customerID) 
        REFERENCES customers(customerID)
);
GO

-- Table: login_logs
CREATE TABLE login_logs (
    id INT IDENTITY(1,1) PRIMARY KEY,
    user_id INT NOT NULL,
    login_time DATETIME NOT NULL DEFAULT GETDATE(),
    CONSTRAINT FK_login_logs_staff FOREIGN KEY (user_id) 
        REFERENCES staff(id)
);
GO

-- Table: store
CREATE TABLE store (
    storeID INT IDENTITY(1,1) PRIMARY KEY,
    name_store VARCHAR(200) NOT NULL,
    location_store VARCHAR(200) NOT NULL,
    contactNo VARCHAR(15) NULL,
    opening_hours VARCHAR(100) NULL,
    created_at DATETIME NOT NULL DEFAULT GETDATE(),
    updated_at DATETIME NOT NULL DEFAULT GETDATE()
);
GO

-- Table: total_expense
CREATE TABLE total_expense (
    id INT IDENTITY(1,1) PRIMARY KEY,
    expenses DECIMAL(10,2) NULL,
    created_at DATETIME NOT NULL DEFAULT GETDATE()
);
GO

-- Table: total_income
CREATE TABLE total_income (
    id INT IDENTITY(1,1) PRIMARY KEY,
    income DECIMAL(10,2) NOT NULL,
    created_at DATETIME NOT NULL DEFAULT GETDATE()
);
GO

-- Table: transactions
CREATE TABLE transactions (
    transaction_id INT IDENTITY(1,1) PRIMARY KEY,
    customerID INT NOT NULL,
    amount DECIMAL(10,2) NOT NULL,
    date DATETIME DEFAULT GETDATE(),
    CONSTRAINT FK_transactions_customers FOREIGN KEY (customerID) 
        REFERENCES customers(customerID) ON DELETE CASCADE
);
GO

-- =============================================
-- Create Triggers
-- =============================================

-- Trigger: update_inventory_supplier (After INSERT on expenses)
CREATE TRIGGER trg_update_inventory_supplier
ON expenses
AFTER INSERT
AS
BEGIN
    SET NOCOUNT ON;
    
    -- Update inventory quantity and stock_supply
    UPDATE i
    SET quantity = i.quantity + ins.order_quantity,
        stock_supply = i.stock_supply - ins.order_quantity
    FROM inventory i
    INNER JOIN inserted ins ON i.id = ins.product_id;
    
    -- Update suppliers stock_supply
    UPDATE s
    SET stock_supply = s.stock_supply - ins.order_quantity
    FROM suppliers s
    INNER JOIN inserted ins ON s.id = (
        SELECT supplier_id FROM inventory WHERE id = ins.product_id
    );
END;
GO

-- Trigger: deduct_inventory_quantity (After INSERT on purchases)
CREATE TRIGGER trg_deduct_inventory_quantity
ON purchases
AFTER INSERT
AS
BEGIN
    SET NOCOUNT ON;
    
    UPDATE inventory
    SET inventory.quantity = inventory.quantity - ins.quantity
    FROM inventory
    INNER JOIN inserted ins ON inventory.id = ins.purchaseID;
END;
GO
